# x17 sub-form table: multi-syllable bundles with a fixed base per syllable.
#
# One line per sub-form.  Whitespace-separated columns:
#   name   syllables   base_head   base_s1   base_s2 [base_s3 ...]
#
# Each `base_*` is the all-operands-zero value of that syllable.  The
# assembler OR's operand field contributions (listed in operands.txt with
# a `syll` column) on top of the base.
#
# All x17.* sub-forms are 3 syllables (head + S1 + S2); the `.halt` case
# has no operand fields, the others have at least one.  Sources: the ref
# strings in isa.toml plus the silicon-measured byte patterns in
# the probe list (own-x17alu-shl, own-x17imm, own-x17ctl, ...).

# x19.imm: 4-syllable VLIW; AGU with a 16-bit immediate.  Class triple
# (000, 000, 001).  Semantics from fact 146 / own-x19imm:
#   e<{w1[0], head[31:29]}> <- AGU10[w1[4:1]](x = e<w3[28:25]>,
#                                             y = R<head[28:25]>,
#                                             i = w2[30:15])
# For the measured gate, fn_op = 10 (y + i); head/w1/w3 bake everything
# except the imm itself (16 bits at w2[30:15]).  dest = e3.
x19.imm    4   0x60000019   0x00060614   0x00002000   0x21F20000

# x19.mul: 4-syllable VLIW; scalar multiply with dual R/e write-back.
# Class triple (000, 001, 110).  Semantics from fact 145 / own-x19mul
# (op = 4): e<w1[6:3]>, R<w3[10:7]> <- low16(s<w2[8:5]> * s<w2[4:1]>).
# w1[17] is "no doubling"; w1[15] and w2[24] are the measured enables.
x19.mul    4   0x00000019   0x00028000   0x21000000   0xC0000000

# agu10r: 2-syllable bundle.  Scalar immediate load + AGU10 into an R reg.
# Semantics from fact 131 / own-agu10r:
#   s<head[24:21]> <- head[20:5]
#   R<S[24:21]> <- AGU10[S[4:1]](x = e<S[28:25]>, y = R<4*head[28:27]>)
# S[20] = 1 required (baked); S[18]/S[17] also measured enables.  S class
# 101 (bits 31, 29 set).  own-agu10r = scalar imm + AGU out of an element.
agu10r     2   0x00000010   0xA0160000

# ldi2: 2-syllable low5 0x10 bundle with S class 001.  Loads an s register
# with a 16-bit imm (fact 74 / own-ldi2):
#   s<head[25:21]> <- head[20:5]     (S[18] = 1 required; baked into base)
ldi2       2   0x00000010   0x20040000

# alu11: 2-syllable bundle.  Head is an ALU05 word, S (class 001) packs
# an AGU10 op in parallel.  Semantics from fact 129 / own-alu11 +
# own-agu11.  Setting an operand to zero disables that half.
#   Head side: s<head[18:15]> <- ALU05[head[10:5]](s<head[14:11]>,
#                                                 s<head[29:26]>)
#   S side (class 001): R<S[28:25]>, e<S[12:9]> <-
#       (S[8] ? AGU10 : AGU10B)[S[20:17]](x = e<S[24:21]>,
#                                         y = R<S[16:13]>)
# S[29] = 1 required (else wedge); baked into base.
alu11      2   0x00000011   0x20000000

# x19.aagu: 4-syllable VLIW; updates one A register with a strided AGU op.
# Class triple (w1[31:29], w2[31:29], w3[31:29]) = (000, 000, 010).
# Semantics from fact 144 / own-x19aagu (fn_op = 2):
#   A<w1[7:3]> <- AGUA[w3[24:20]](x = A<w3[19:15]>, i = w3[14:0])
# For fn_op 2 the formula is x + 2*i.  w1[26] is the measured enable.
x19.aagu   4   0x00000019   0x06000000   0x00000000   0x40000000

# x19.ld: 4-syllable VLIW doing load + AGU + store together.  Class triple
# (w1[31:29], w2[31:29], w3[31:29]) = (000, 000, 111).  Semantics from
# fact 143 / own-x19ld:
#   s<w2[13:10]> <- Y:[e<w3[28:25]>]
#   e<2*w1[2:0] + head[31]> <- AGU10[w1[10:7]](x = e<src>, y = e<src>)
#   st.y Y:[e<new>] <- v[cur_lane]
# w1[3] = 1 is the measured enable; w3 bakes class 111 + the low 12 bits
# measured on the gate.
x19.ld     4   0x00000019   0x00000008   0x00000000   0xE0000403

# x17.cont: control bundle that lets execution fall through.  S2 class 101
# + S1[22:20] = 000.  Semantics from fact 133 / own-x17ctl:
#   (no state change; the PC advances to the next bundle)
# S1 base pins the class-001 + S1[28] + S1[15] enables as measured.
x17.cont   3   0x00000017   0x31008000   0xA0000000

# x17.halt: canonical stop bundle.  Head carries only the x17 opcode;
# S1 = 0x31308000 is the standard "halt S1" (S1 class 001, S1[15]=1, ...);
# S2 class 101 (bits [31:29] = 101).
x17.halt   3   0x00000017   0x31308000   0xA0000000

# x17.imm: scalar immediate load into s<S1[10:7]> from head[26:11].
# S1 class 011 ([31:29]=011) + S2 class 111 ([31:29]=111); S1[14]=1 is the
# "enable" bit documented in isa.toml (needed, else wedge).
x17.imm    3   0x00000017   0x60004000   0xE0000000

# x17.xld_a10: X-memory load with A-reg offset from S1 class bits.  S1
# class 010 + S2 class 110, head[27] clear.  Semantics from fact
# 141 / own-x17xld-a10:
#   s<S2[13:10]> <- X:[A<8*S2[1:0] + S1[31:29]>]      (S1[17:15] = 101)
# This sub-form pins S1 class = 010 (A-offset = 2); other offsets await
# their own measurement.  S1 is baked whole into the base.
x17.xld_a10 3   0x00000017   0x4346A500   0xC0204000

# x17.xpm: X-memory load into a scalar, continuing bundle.  S1 class 100
# + S2 class 110.  Semantics from fact 139 / own-x17xpm:
#   s<S2[13:10]> <- X:[A<8*S2[1:0] + 4>]              (S1[17:15] = 101)
# The A<S1[25:22]> write (secondary A-reg store) is OPEN today - S1 is
# baked whole into the base, exposing only the load dest + a-half picker.
x17.xpm    3   0x08000017   0x8346A500   0xC0204000

# x17.rsub: reverse-subtract with immediate b, continuing bundle.  S1 class
# 000 + S2 class 001.  Semantics from fact 140 / own-x17rsub (op = 5):
#   s<S1[22:19]> <- (S1[11:8]) - s<S2[28:25]>      (24-bit wrap)
# S1[25:23] = 001 selects the imm-b mode; S1[26] is the measured enable.  S1[14:12] op selector and S1[7:4] predicate slot are passed
# through; today only op = 5 is modelled (b - a).
x17.rsub   3   0x00000017   0x04800000   0x20000000

# x17.mve: move scalar + 1 into an element, continuing bundle.  S1 class
# 000 + S2 class 010.  Semantics from fact 142 / own-x17mve:
#   e<S1[19:16]> <- s<S2[28:25]> + 1    (S1[9] enable)
# Head bit 27 is the sub-form marker; head[10] clear.  S1 base bakes bits
# 10, 9 (enable pair).  S2 base is the class-010 marker only.
x17.mve    3   0x08000017   0x00000600   0x40000000

# x17.alu05: head is an ALU05 word, continuing bundle.  S1 class 000 +
# S2 class 110.  Semantics from fact 138 / own-x17alu05:
#   s<head[18:15]> <- ALU05[head[10:5]](a = s<head[14:11]>, b = s<S2[28:25]>)
# S1[13]=1 is the measured enable; S2 class 110 only.  Head base keeps the
# low5=0x17 marker and lets dest/a/op OR on top.
x17.alu05  3   0x00000017   0x00002000   0xC0000000

# x17.ld: scalar asr + Y-memory load into element, continuing bundle.
# S1 class 000 + S2 class 011.  Semantics from fact 137 / own-x17ld:
#   s<S1[22:19]> <- s<S2[28:25]> asr S1[26:23]
#   e<2*S1[2:0]> <- Y:[e<S1[18:15]>]
# Head bits [6:5]=11 (op=3) baked into base = 0x77; S1 base = 0x08000000
# (bit 27 enable); S2 base = 0x60001208 (class 011 + measured enables).
x17.ld     3   0x00000077   0x08000000   0x60001208

# x17.e: dual element write in a continuing bundle.  S1 class 000 + S2
# class 100.  Semantics from fact 135 / own-x17e:
#   e<S2[11:8]>, s<head[24:21]> <- head[20:5]   (gated by S1[18])
#   e<S1[16:13]> <- old e<S2[19:16]> - 1        (gated by S1[17])
# Base S1 = 0x00060200 bakes both enables and S1[9]=1 measured on
# own-x17e; base S2 = 0x80000000 (class 100 only).
x17.e      3   0x00000017   0x00060200   0x80000000

# x17.agu: address-generation op in a continuing bundle.  S1 class 001
# (with S1[28] = 1) + S2 class 010.  Semantics from upstream emu.py:
#   dest_R <- AGU10[fn_op](x = e<S1[8:5]>, y = R<S2[4:1]>)
# Base S1 = 0x30000000 (class 001 + S1[28]=1); base S2 = 0x40000000
# (class 010 only; S2[4:1] is the y operand, S2[8:5] is dest).
x17.agu    3   0x00000017   0x30000000   0x40000000

# x17.alu: ALU op in a continuing bundle.  S1 class 001 + S2 class 111:
#   s<S1[27:24]> <- X17OP[head[10:7]](a = s<S2[28:25]>, b = s<S1[19:16]>, n = S1[23:20])
# Base S1 = 0x30008000: class bits [31:29]=001 plus S1[28]=1, S1[15]=1 (the
# enable marker).  Base S2 = 0xE0000000 (class 111 only; a goes into [28:25]).
# Silicon-measured: own-x17alu-shl (op=0 shl, n=3) and own-x17alu-rsub
# (op=4 b-a) both match this base.
x17.alu    3   0x00000017   0x30008000   0xE0000000
